# 自身のAS番号を 202734 と仮定

define MY_ASN = 202734;
define MY_PREFIXES = [
    192.0.2.0/24+,
    203.0.113.0/24+
];

# 自身がオリジンである経路のみを許可する関数
function is_my_origin() {
    # AS PATHの末尾（オリジン）が自身のASかチェック
    if bgp_path.last = MY_ASN then return true;
    return false;
}

# コレクター向けBGPプロトコル設定
protocol bgp collector_he {
    local as MY_ASN;
    neighbor 2001:470::1 as 6939; # HE Route Collector (例)
    
    ipv6 {
        import none; # コレクターからは経路を受け取らない
        export filter {
            # 自分が所有するプレフィックスかつ、自身のAS起源のみ許可
            if (net ~ MY_PREFIXES) && is_my_origin() then {
                accept;
            }
            # それ以外は暗黙的・明示的にすべてドロップ
            reject;
        };
    };
}
```

`export all` という安易な設定やフィルタの書き忘れは、最悪の場合インターネット全体に影響を及ぼす事故の元です。

## 考察

今回の騒動から、インフラエンジニアやネットワークエンジニアが学ぶべき示唆は2つあります。

第一に、「可視化ツールの結果を盲信せず、事象を正しく解釈する」ことです。HE BGP Toolkitなどのツールは非常に有用ですが、それはあくまで「特定の観測点（コレクター）から送られてきたコントロールプレーンのスナップショット」に過ぎません。ハイジャックの疑いがある場合は、単一のツールだけでなく、RIPE statなどの複数観測点からのDFZの状況確認や、データプレーンでの実際の疎通確認（ping/traceroute）を併用し、冷静に影響範囲を見極めるべきです。

第二に、「現代のインターネットルーティングは想定以上に堅牢に進化している」というポジティブな事実です。10年前であれば、このような一介のASの設定ミス（Bogonリーク）がトランジットを越えて伝播し、かつての「パキスタンによるYouTubeハイジャック事件（2008年）」のような世界的な大惨事になっていた可能性があります。しかし今回は、周辺のISPが適切にRPKI ROA（Route Origin Authorization）やIRRベースのフィルタリングを実装していたため、インターネットの自浄作用が働き被害はゼロに抑えられました。

インフラエンジニアとしては、自組織の経路をRPKIで保護（ROAの作成）すること、そしてアップストリームやピアに対してROV（Route Origin Validation）に基づくフィルタリングを適用することの重要性が、実戦で証明されたケースと言えるでしょう。

## まとめ

AS202734の「4,000プレフィックスハイジャック騒動」は、悪意あるサイバー攻撃ではなく、ルートコレクターへのExportフィルタの不備という単純なミスが引き起こした「見せかけのインシデント」でした。

しかし、このミスが実害を生まなかったのは、インターネットコミュニティ全体で長年推進されてきたルーティングセキュリティ（RPKIやMANRSの取り組み）の努力の賜物です。私たちは、ツールが発するアラートの裏側にある「ネットワークの実際の動き」を正確に読み解く力を養うとともに、自組織のルーター設定において「出さない・受け取らない」確実なフィルタリング設定を引き続き徹底していく必要があります。
